\includegraphics[width=0.84\linewidth]{figures/supp/supp_landscape_error_kappa.pdf}
\caption{\textbf{Error landscapes: deterministic equivalent versus
natural-image Monte Carlo} over the full grid of Fig.~\ref{fig:pixel}
(114 noise levels $\times$ 478 penalties, $\kappa$ up to $10^4$).
\textbf{A,\,B}~Leading DE of the normalized prediction error $E_{gen}/S$ and

\caption{\textbf{$R^2$ landscapes}, layout as in

$R^2_{acc}<-10^{4}$ at weak regularization; the prediction-optimal path meets
the wedge at $\sigma^2/S\approx1$ (Fig.~\ref{fig:pixel}).}

(left) and $\mathrm{slope}_{acc}=N/D$ for accentuation (right); color is

Strong shrinkage makes both slopes exceed one.  Below the prediction-optimal

the teacher far less than the student predicts.}
\label{fig:supp-pixel-slope}
\end{figure}

\begin{figure}[tbp]
\centering
% Source: CR/figures/vanhateren_landscape/supp/supp_landscape_error_lam.pdf
\includegraphics[width=0.84\linewidth]{figures/supp/supp_landscape_error_lam.pdf}
\caption{\textbf{Error landscapes against the ridge penalty $\lambda$}
($\alpha=n\lambda$) instead of the effective penalty $\kappa$; data, panels
and lines as in Fig.~\ref{fig:supp-pixel-de-mc}.  Because $d>n$, all
$\lambda\lesssim10^{-4}$ map onto the near-ridgeless floor of $\kappa$, so
this view resolves the low-noise corner that the $\kappa$ view compresses.
There the prediction optimum is ridgeless ($\lambda^\star_{gen}$ at the grid
floor for $\sigma^2/S\lesssim5\times10^{-5}$), whereas the control optimum
$\lambda^\star_{acc}$ shrinks in proportion to the noise along a deep valley of
$E_{acc}$: at $\sigma^2/S=10^{-6}$ the two optima differ 4000-fold in
$\lambda$ but only by 3\% in $\kappa$.  For $\sigma^2/S\ge10^{-3}$,
$\lambda^\star_{acc}\approx2$--$2.5\,\lambda^\star_{gen}$.}
\label{fig:supp-pixel-lambda}
\end{figure}
